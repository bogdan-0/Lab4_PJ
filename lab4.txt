public class Imprimanta extends Echipamente
{
    private int ppm;
    private int rezolutie;
    private int p_car;
    private ModTiparire mod;

    public Imprimanta(int ppm, int rezolutie, int p_car, ModTiparire mod) {
        this.ppm = ppm;
        this.rezolutie = rezolutie;
        this.p_car = p_car;
        this.mod = mod;
    }

    public int getPpm() {
        return ppm;
    }

    public void setPpm(int ppm) {
        this.ppm = ppm;
    }

    public int getRezolutie() {
        return rezolutie;
    }

    public void setRezolutie(int rezolutie) {
        this.rezolutie = rezolutie;
    }

    public int getP_car() {
        return p_car;
    }

    public void setP_car(int p_car) {
        this.p_car = p_car;
    }

    public ModTiparire getMod() {
        return mod;
    }

    public void setMod(ModTiparire mod) {
        this.mod = mod;
    }

    @java.lang.Override
    public java.lang.String toString() {
        return "Imprimanta{" +
                "ppm=" + ppm +
                ", rezolutie=" + rezolutie +
                ", p_car=" + p_car +
                ", mod=" + mod +
                '}';
    }
}

public enum FormatCopiere
{   A3,
    A4
}


public abstract class Echipamente
{
    protected String denumire;
    protected int nr_inv;
    protected double pret;
    protected String zona_magazin;
    protected Stare stare;

    public Echipamente(String denumire, int nr_inv, double pret, String zona_magazin, Stare stare) {
        this.denumire = denumire;
        this.nr_inv = nr_inv;
        this.pret = pret;
        this.zona_magazin = zona_magazin;
        this.stare = stare;
    }

    public String getDenumire() {
        return denumire;
    }

    public void setDenumire(String denumire) {
        this.denumire = denumire;
    }

    public int getNr_inv() {
        return nr_inv;
    }

    public void setNr_inv(int nr_inv) {
        this.nr_inv = nr_inv;
    }

    public double getPret() {
        return pret;
    }

    public void setPret(double pret) {
        this.pret = pret;
    }

    public String getZona_magazin() {
        return zona_magazin;
    }

    public void setZona_magazin(String zona_magazin) {
        this.zona_magazin = zona_magazin;
    }

    public Stare getStare() {
        return stare;
    }

    public void setStare(Stare stare) {
        this.stare = stare;
    }

    @java.lang.Override
    public java.lang.String toString() {
        return "Echipamente{" +
                "denumire='" + denumire + '\'' +
                ", nr_inv=" + nr_inv +
                ", pret=" + pret +
                ", zona_magazin='" + zona_magazin + '\'' +
                ", stare=" + stare +
                '}';
    }
}
